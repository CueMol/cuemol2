# module loader generated by MCWG
SET(RENDER_MODLDR_SRC render.moddef)

MCWRAPGEN_CLASS(RENDER_SRCS ${RENDER_MCWG_QIFS})
MCWRAPGEN_MODULE(RENDER_SRCS ${RENDER_MODLDR_SRC} ${RENDER_MCWG_QIFS})

# add_library(render SHARED ${RENDER_SRCS} ${MCWG_HEADERS})
add_library(render STATIC ${RENDER_SRCS} ${MCWG_HEADERS})

if (WIN32)
  # add_definitions(-DRENDER_EXPORTS=1)
  target_compile_definitions(render PUBLIC RENDER_EXPORTS=1)
else()
  # target_compile_options(render PUBLIC "-fPIC")
  set_property(TARGET render PROPERTY POSITION_INDEPENDENT_CODE ON)
endif()

if (USE_PCH)
  target_precompile_headers(render PRIVATE ${CMAKE_SOURCE_DIR}/src/common.h)
endif ()

target_link_libraries(render PRIVATE qlib gfx qsys CGAL::CGAL qmpng qmzlib Boost::filesystem)

# umbreon (Embree ray-tracer) backend. Only linked when ENABLE_UMBREON is on;
# the Umbreon{DisplayContext,SceneExporter} sources are always compiled but
# guard their umbreon usage with HAVE_UMBREON, so the default build links no
# umbreon symbols.
if (ENABLE_UMBREON)
  target_link_libraries(render PRIVATE umbreon::umbreon)
endif ()

MCWRAPGEN_SCR_WRAPPERS(render)
